cmake_minimum_required(VERSION 3.22)
include(cmake/Version.cmake)
project(Archcodex VERSION ${ARCHCODEX_NUMERIC_VERSION} LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
include(GNUInstallDirs)
include(CTest)
option(ARCHCODEX_GUI "Build the Qt/KDE desktop application" ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Network Sql)
find_package(Boost 1.74 REQUIRED)
find_package(Threads REQUIRED)

add_library(archcodex-core src/types.cpp src/store.cpp src/backend.cpp src/localbackend.cpp
  src/discovery.cpp src/discovery.qrc src/launch.cpp src/worker.cpp src/asyncbackend.cpp
  src/asyncstore.cpp src/containerdiscovery.cpp src/jsonprocess.cpp src/workspace.cpp
  src/opening.cpp src/updater.cpp)
target_include_directories(archcodex-core PUBLIC src)
target_link_libraries(archcodex-core PUBLIC Qt6::Core Qt6::Network Qt6::Sql)
target_compile_definitions(archcodex-core PUBLIC ARCHCODEX_VERSION="${ARCHCODEX_VERSION}")
target_compile_options(archcodex-core PRIVATE -Wall -Wextra -Wpedantic)

add_executable(archcodex-bridge src/bridge.cpp)
set_target_properties(archcodex-bridge PROPERTIES AUTOMOC OFF)
target_include_directories(archcodex-bridge PRIVATE ${Boost_INCLUDE_DIRS})
target_link_libraries(archcodex-bridge PRIVATE Threads::Threads)
target_compile_definitions(archcodex-bridge PRIVATE BOOST_ERROR_CODE_HEADER_ONLY)
target_compile_options(archcodex-bridge PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS archcodex-bridge RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})

if(ARCHCODEX_GUI)
  find_package(Qt6 REQUIRED COMPONENTS Widgets DBus)
  find_package(KF6CoreAddons REQUIRED)
  find_package(KF6Parts REQUIRED)
  find_package(KF6XmlGui REQUIRED)
  find_package(KF6ConfigWidgets REQUIRED)
  find_package(KF6WidgetsAddons REQUIRED)
  find_package(KF6Notifications REQUIRED)
  find_package(KF6WindowSystem REQUIRED)
  find_path(TERMINAL_INTERFACE_INCLUDE_DIR kde_terminal_interface.h PATH_SUFFIXES KF6/KParts KF6 REQUIRED)
  add_executable(archcodex src/main.cpp src/window.cpp src/commands.cpp src/settingsdialog.cpp src/discoverydialog.cpp src/sessionmodel.cpp src/terminal.cpp src/notifications.cpp src/updatedialog.cpp)
  target_include_directories(archcodex PRIVATE ${TERMINAL_INTERFACE_INCLUDE_DIR})
  target_link_libraries(archcodex PRIVATE archcodex-core Qt6::Widgets Qt6::DBus KF6::CoreAddons KF6::Parts KF6::XmlGui KF6::ConfigWidgets KF6::WidgetsAddons KF6::Notifications KF6::WindowSystem)
  target_compile_options(archcodex PRIVATE -Wall -Wextra -Wpedantic)
  install(TARGETS archcodex RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
  install(FILES data/io.github.emsi.Archcodex.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
  configure_file(data/io.github.emsi.Archcodex.metainfo.xml.in
    io.github.emsi.Archcodex.metainfo.xml @ONLY)
  install(FILES ${CMAKE_CURRENT_BINARY_DIR}/io.github.emsi.Archcodex.metainfo.xml
    DESTINATION ${CMAKE_INSTALL_DATADIR}/metainfo)
  install(FILES data/io.github.emsi.Archcodex.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps)
  install(FILES data/archcodex.notifyrc DESTINATION ${CMAKE_INSTALL_DATADIR}/knotifications6)
endif()
install(FILES LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/archcodex)
